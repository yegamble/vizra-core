## The shape at main (PR#7 VERIFY FINDING 1), reproduced verbatim:

          - name: Assert the toolchain is absent from the runtime image
            run: |
              set -euo pipefail
              # The list is scripts/runtime-toolchain-denylist.txt; that file says
              # why it is a file and not an inline list.
              tools=$(grep -v '^#' scripts/runtime-toolchain-denylist.txt | grep -v '^$' | tr '\n' ' ')
              echo "checking for: $tools"
              present=$(docker run --rm -e TOOLS="$tools" vizra-core:ci sh -c '
                for t in $TOOLS; do
                  command -v "$t" >/dev/null 2>&1 && echo "$t"
                done
                true') || true
              if [ -n "$present" ]; then
                echo "::error::the runtime image carries build tooling:"
                printf '%s\n' "$present"
                exit 1
              fi
              echo "no build tooling on PATH"
    
              devpkgs=$(docker run --rm vizra-core:ci sh -c "dpkg-query -W -f='\${binary:Package}\n' 2>/dev/null | grep -- '-dev$' || true")
              if [ -n "$devpkgs" ]; then
                echo "::error::the runtime image carries -dev packages:"
                printf '%s\n' "$devpkgs"
                exit 1
              fi
              echo "no -dev packages installed"
    
              if docker run --rm vizra-core:ci test -e /src; then
                echo "::error::the libvips source tree is still in the runtime image"
                exit 1
              fi
              echo "no /src"

## The new script, same stub:
  clean          EXIT=0  assert-runtime-image: ok — vizra-core:ci carries no toolchain, no -dev packages and no source tree
  broken         EXIT=1  ::error::assert-runtime-image: the container did not run while checking the build toolchain (docker exited 125).
  has-toolchain  EXIT=1  ::error::assert-runtime-image: the runtime image carries build tooling:
  has-src        EXIT=1  ::error::assert-runtime-image: the libvips source tree is still in the runtime image (/src exists)
  weird-exit     EXIT=1  ::error::assert-runtime-image: the container did not run while checking /src (docker exited 7).

  'broken' is the FINDING 1 case: a daemon that cannot run anything. The old shape
  printed 'no build tooling on PATH' and passed; the new one is RED and says so.
